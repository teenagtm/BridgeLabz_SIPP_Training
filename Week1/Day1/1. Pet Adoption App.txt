1. Pet Adoption App
CLASS Pet
    ATTRIBUTES: name, age
    METHOD displayInfo()
        PRINT name, age

CLASS Dog EXTENDS Pet
    ATTRIBUTES: breed
    METHOD displayInfo()
        CALL super.displayInfo()
        PRINT breed

CLASS Cat EXTENDS Pet
    ATTRIBUTES: isIndoor
    METHOD displayInfo()
        CALL super.displayInfo()
        PRINT isIndoor

CLASS Bird EXTENDS Pet
    ATTRIBUTES: canFly
    METHOD displayInfo()
        CALL super.displayInfo()
        PRINT canFly

MAIN
    CREATE dog = Dog("Bruno", 5, "Labrador")
    CREATE cat = Cat("Kitty", 3, true)
    CREATE bird = Bird("Tweety", 2, false)

    CALL dog.displayInfo()
    CALL cat.displayInfo()
    CALL bird.displayInfo()
